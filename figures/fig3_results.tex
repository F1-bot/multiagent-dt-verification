% confidence band between y-ci and y+ci of a table column
\newcommand{\band}[4]{% #1 file, #2 x col, #3 y col, #4 colour
  \addplot[name path=up, draw=none, forget plot] table[x=#2, y expr=\thisrow{#3}+\thisrow{#3ci}] {#1};
  \addplot[name path=dn, draw=none, forget plot] table[x=#2, y expr=\thisrow{#3}-\thisrow{#3ci}] {#1};
  \addplot[#4, forget plot] fill between[of=up and dn];}
\newcommand{\bandpct}[4]{% the same in per cent
  \addplot[name path=up, draw=none, forget plot] table[x=#2, y expr=100*(\thisrow{#3}+\thisrow{#3ci})] {#1};
  \addplot[name path=dn, draw=none, forget plot] table[x=#2, y expr=100*(\thisrow{#3}-\thisrow{#3ci})] {#1};
  \addplot[#4, forget plot] fill between[of=up and dn];}
